cmake_minimum_required(VERSION 3.22)
project(homework_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_STRESS_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
if(NOT HW_SOURCE OR NOT HW_KIND MATCHES "^(combined|ttl|metadata|baseline)$")
  message(FATAL_ERROR "Set HW_SOURCE and HW_KIND=combined|ttl|metadata|baseline")
endif()
add_subdirectory("${HW_SOURCE}" student EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_executable(homework_bench common/main.cc)
target_include_directories(homework_bench PRIVATE "${HW_SOURCE}" "${CMAKE_BINARY_DIR}/student/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(homework_bench PRIVATE LEVELDB_PLATFORM_POSIX=1 LEVELDB_HAS_PORT_CONFIG_H=1)
target_link_libraries(homework_bench PRIVATE leveldb Threads::Threads)
target_compile_options(homework_bench PRIVATE -fno-rtti)
if(HW_KIND STREQUAL "combined")
  target_compile_definitions(homework_bench PRIVATE HW_TTL=1 HW_METADATA=1 HW_COMBINED=1)
elseif(HW_KIND STREQUAL "ttl")
  target_compile_definitions(homework_bench PRIVATE HW_TTL=1)
elseif(HW_KIND STREQUAL "metadata")
  target_compile_definitions(homework_bench PRIVATE HW_METADATA=1)
endif()
if(HW_STORAGE_ADAPTER)
  target_compile_definitions(homework_bench PRIVATE HW_STORAGE_ADAPTER="${HW_STORAGE_ADAPTER}")
endif()
if(HW_TEACHER_DIR)
  target_include_directories(homework_bench PRIVATE "${HW_TEACHER_DIR}")
  target_compile_definitions(homework_bench PRIVATE HW_TEACHER=1)
endif()
# Regression source comes from the pinned assignment baseline, not the submission.
if(HW_REGRESSION_SOURCE)
  file(GLOB reg_db "${HW_REGRESSION_SOURCE}/db/*_test.cc")
  file(GLOB reg_util "${HW_REGRESSION_SOURCE}/util/*_test.cc")
  list(FILTER reg_db EXCLUDE REGEX "/fault_injection_test.cc$")
  list(FILTER reg_util EXCLUDE REGEX "/env(_posix|_windows)?_test.cc$")
  add_executable(homework_regression ${reg_db} ${reg_util}
    "${HW_REGRESSION_SOURCE}/helpers/memenv/memenv_test.cc"
    "${HW_REGRESSION_SOURCE}/table/filter_block_test.cc"
    "${HW_REGRESSION_SOURCE}/table/table_test.cc"
    "${HW_REGRESSION_SOURCE}/util/testutil.cc")
  target_include_directories(homework_regression PRIVATE "${HW_SOURCE}" "${CMAKE_BINARY_DIR}/student/include")
  target_compile_definitions(homework_regression PRIVATE LEVELDB_PLATFORM_POSIX=1 LEVELDB_HAS_PORT_CONFIG_H=1)
  target_link_libraries(homework_regression PRIVATE leveldb gmock gtest_main Threads::Threads)
  target_compile_options(homework_regression PRIVATE -fno-rtti)
  foreach(pair "homework_c_regression|db/c_test.c" "homework_env_regression|util/env_posix_test.cc")
    string(REPLACE "|" ";" fields "${pair}")
    list(GET fields 0 name)
    list(GET fields 1 source)
    add_executable(${name} "${HW_REGRESSION_SOURCE}/${source}" "${HW_REGRESSION_SOURCE}/util/testutil.cc")
    target_include_directories(${name} PRIVATE "${HW_SOURCE}" "${CMAKE_BINARY_DIR}/student/include")
    target_compile_definitions(${name} PRIVATE LEVELDB_PLATFORM_POSIX=1 LEVELDB_HAS_PORT_CONFIG_H=1)
    target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>)
    target_link_libraries(${name} PRIVATE leveldb gmock gtest Threads::Threads)
  endforeach()
endif()
